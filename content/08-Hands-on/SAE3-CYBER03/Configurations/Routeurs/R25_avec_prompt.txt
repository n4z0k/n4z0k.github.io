Router>enable
Router#configure terminal
Router(config)#hostname R25
R25(config)#no ip domain-lookup
R25(config)#service password-encryption
R25(config)#enable secret Sae3Cyber!
R25(config)#username admin privilege 15 secret Sae3Cyber!
R25(config)#banner motd #Acces reserve - SAE3-CYBER03 - R25#
R25(config)#ip domain-name sae3cyber03.iut
!
! ---------- QoS (methode du TP4) : etape 1 - classification et marquage en entree ----------
! Flux video (TCP 8080) entre chaque serveur et les clients publics du site d'en face
R25(config)#ip cef
R25(config)#ip access-list extended VIDEO
R25(config-ext-nacl)#permit tcp host 50.50.25.1 eq 8080 150.150.26.0 0.0.0.255
R25(config-ext-nacl)#permit tcp host 50.50.26.1 eq 8080 150.150.25.0 0.0.0.255
R25(config-ext-nacl)#permit tcp 150.150.26.0 0.0.0.255 host 50.50.25.1 eq 8080
R25(config-ext-nacl)#permit tcp 150.150.25.0 0.0.0.255 host 50.50.26.1 eq 8080
R25(config-ext-nacl)#exit
R25(config)#class-map match-any VIDEO
R25(config-cmap)#match access-group name VIDEO
R25(config-cmap)#exit
R25(config)#policy-map MARQUAGE
R25(config-pmap)#class VIDEO
R25(config-pmap-c)#set dscp af41
R25(config-pmap-c)#exit
R25(config-pmap)#exit
!
! ---------- QoS (methode du TP4) : etape 2 - politique de sortie ----------
! Les flux video sont reconnus par leur marquage DSCP AF41 (34), pose a l'entree du reseau
R25(config)#class-map match-all DSCP-VIDEO
R25(config-cmap)#match dscp af41
R25(config-cmap)#exit
R25(config)#policy-map QOS-VIDEO
R25(config-pmap)#class DSCP-VIDEO
R25(config-pmap-c)#bandwidth percent 30
R25(config-pmap-c)#exit
R25(config-pmap)#class class-default
R25(config-pmap-c)#fair-queue
R25(config-pmap-c)#exit
R25(config-pmap)#exit
!
! ---------- ACL : protection du reseau des serveurs publics ----------
! Appliquee en SORTIE de l'interface serveurs : seul le trafic destine aux serveurs est examine
R25(config)#ip access-list extended PROTECT-SERVEURS
R25(config-ext-nacl)#remark Web
R25(config-ext-nacl)#permit tcp any host 50.50.25.1 eq www
R25(config-ext-nacl)#permit tcp any host 50.50.25.1 eq 443
R25(config-ext-nacl)#remark SSH
R25(config-ext-nacl)#permit tcp any host 50.50.25.1 eq 22
R25(config-ext-nacl)#remark FTP (controle, actif, passif 40000-40100)
R25(config-ext-nacl)#permit tcp any host 50.50.25.1 eq ftp
R25(config-ext-nacl)#permit tcp any host 50.50.25.1 eq ftp-data
R25(config-ext-nacl)#permit tcp any host 50.50.25.1 range 40000 40100
R25(config-ext-nacl)#remark Video : uniquement clients publics du site d'en face
R25(config-ext-nacl)#permit tcp 150.150.26.0 0.0.0.255 host 50.50.25.1 eq 8080
R25(config-ext-nacl)#remark DHCP relaye par le routeur
R25(config-ext-nacl)#permit udp any host 50.50.25.1 eq bootps
R25(config-ext-nacl)#remark ICMP pour les tests
R25(config-ext-nacl)#permit icmp any host 50.50.25.1 echo
R25(config-ext-nacl)#permit icmp any host 50.50.25.1 echo-reply
R25(config-ext-nacl)#permit icmp any host 50.50.25.1 unreachable
R25(config-ext-nacl)#deny ip any any log
R25(config-ext-nacl)#exit
!
! ---------- ACL : aucune adresse privee ne doit sortir vers l'ISR ----------
! Evaluee apres le NAT : seules les adresses deja traduites passent
R25(config)#ip access-list extended FILTRE-PRIVE
R25(config-ext-nacl)#deny ip 10.0.0.0 0.255.255.255 any
R25(config-ext-nacl)#deny ip 172.16.0.0 0.15.255.255 any
R25(config-ext-nacl)#deny ip 192.168.0.0 0.0.255.255 any
R25(config-ext-nacl)#permit ip any any
R25(config-ext-nacl)#exit
!
! ---------- NAT dynamique : 63 premieres adresses privees ----------
R25(config)#ip access-list standard NAT-PRIVE-25
R25(config-std-nacl)#permit 192.168.25.0 0.0.0.63
R25(config-std-nacl)#exit
R25(config)#ip nat pool POOL-NAT-25 90.0.25.2 90.0.25.64 netmask 255.255.255.0
R25(config)#ip nat inside source list NAT-PRIVE-25 pool POOL-NAT-25
!
! ---------- Interfaces ----------
R25(config)#interface FastEthernet0/1
R25(config-if)#description Interco vers ISR25 Ge0/0/1 - 90.0.25.0/24
R25(config-if)#ip address 90.0.25.1 255.255.255.0
R25(config-if)#ip nat outside
R25(config-if)#ip access-group FILTRE-PRIVE out
R25(config-if)#service-policy output QOS-VIDEO
R25(config-if)#no shutdown
R25(config-if)#exit
R25(config)#interface FastEthernet0/0
R25(config-if)#description Trunk 802.1Q vers SW25 port 24
R25(config-if)#no ip address
R25(config-if)#service-policy input MARQUAGE
R25(config-if)#service-policy output QOS-VIDEO
R25(config-if)#no shutdown
R25(config-if)#exit
R25(config)#interface FastEthernet0/0.50
R25(config-subif)#description VLAN 50 - Serveurs publics 50.50.25.0/24
R25(config-subif)#encapsulation dot1Q 50
R25(config-subif)#ip address 50.50.25.254 255.255.255.0
R25(config-subif)#ip access-group PROTECT-SERVEURS out
R25(config-subif)#exit
R25(config)#interface FastEthernet0/0.150
R25(config-subif)#description VLAN 150 - Clients publics 150.150.25.0/24
R25(config-subif)#encapsulation dot1Q 150
R25(config-subif)#ip address 150.150.25.254 255.255.255.0
R25(config-subif)#exit
R25(config)#interface FastEthernet0/0.192
R25(config-subif)#description VLAN 192 - Clients prives 192.168.25.0/24
R25(config-subif)#encapsulation dot1Q 192
R25(config-subif)#ip address 192.168.25.254 255.255.255.0
R25(config-subif)#ip nat inside
R25(config-subif)#ip helper-address 50.50.25.1
R25(config-subif)#exit
!
! ---------- OSPF : zone 25 (zone du site) ----------
! Le reseau prive 192.168.25.0/24 n'est volontairement PAS annonce
R25(config)#router ospf 1
R25(config-router)#router-id 25.0.0.2
R25(config-router)#passive-interface FastEthernet0/0.50
R25(config-router)#passive-interface FastEthernet0/0.150
R25(config-router)#passive-interface FastEthernet0/0.192
R25(config-router)#network 90.0.25.0 0.0.0.255 area 25
R25(config-router)#network 50.50.25.0 0.0.0.255 area 25
R25(config-router)#network 150.150.25.0 0.0.0.255 area 25
