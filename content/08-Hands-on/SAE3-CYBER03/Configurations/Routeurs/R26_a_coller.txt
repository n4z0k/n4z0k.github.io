! ---------- QoS (methode du TP4) : etape 1 - classification et marquage en entree ----------
! Flux video (TCP 8080) entre chaque serveur et les clients publics du site d'en face
ip cef
ip access-list extended VIDEO
 permit tcp host 50.50.25.1 eq 8080 150.150.26.0 0.0.0.255
 permit tcp host 50.50.26.1 eq 8080 150.150.25.0 0.0.0.255
 permit tcp 150.150.26.0 0.0.0.255 host 50.50.25.1 eq 8080
 permit tcp 150.150.25.0 0.0.0.255 host 50.50.26.1 eq 8080
 exit
class-map match-any VIDEO
 match access-group name VIDEO
 exit
policy-map MARQUAGE
 class VIDEO
  set dscp af41
  exit
 exit
!
! ---------- QoS (methode du TP4) : etape 2 - politique de sortie ----------
! Les flux video sont reconnus par leur marquage DSCP AF41 (34), pose a l'entree du reseau
class-map match-all DSCP-VIDEO
 match dscp af41
 exit
policy-map QOS-VIDEO
 class DSCP-VIDEO
  bandwidth percent 30
  exit
 class class-default
  fair-queue
  exit
 exit
!
! ---------- ACL : protection du reseau des serveurs publics ----------
! Appliquee en SORTIE de l'interface serveurs : seul le trafic destine aux serveurs est examine
ip access-list extended PROTECT-SERVEURS
 remark Web
 permit tcp any host 50.50.26.1 eq www
 permit tcp any host 50.50.26.1 eq 443
 remark SSH
 permit tcp any host 50.50.26.1 eq 22
 remark FTP (controle, actif, passif 40000-40100)
 permit tcp any host 50.50.26.1 eq ftp
 permit tcp any host 50.50.26.1 eq ftp-data
 permit tcp any host 50.50.26.1 range 40000 40100
 remark Video : uniquement clients publics du site d'en face
 permit tcp 150.150.25.0 0.0.0.255 host 50.50.26.1 eq 8080
 remark RADIUS depuis le switch d'acces SW26-2 (NAS 802.1X)
 permit udp host 150.150.26.250 host 50.50.26.1 range 1812 1813
 remark DHCP relaye par le routeur
 permit udp any host 50.50.26.1 eq bootps
 remark ICMP pour les tests
 permit icmp any host 50.50.26.1 echo
 permit icmp any host 50.50.26.1 echo-reply
 permit icmp any host 50.50.26.1 unreachable
 deny ip any any log
 exit
!
! ---------- ACL : aucune adresse privee ne doit sortir vers l'ISR ----------
! Evaluee apres le NAT : seules les adresses deja traduites passent
ip access-list extended FILTRE-PRIVE
 deny ip 10.0.0.0 0.255.255.255 any
 deny ip 172.16.0.0 0.15.255.255 any
 deny ip 192.168.0.0 0.0.255.255 any
 permit ip any any
 exit
!
! ---------- NAT dynamique : 63 premieres adresses privees ----------
ip access-list standard NAT-PRIVE-26
 permit 192.168.26.0 0.0.0.63
 exit
ip nat pool POOL-NAT-26 90.0.26.2 90.0.26.64 netmask 255.255.255.0
ip nat inside source list NAT-PRIVE-26 pool POOL-NAT-26
!
! ---------- Interfaces ----------
interface FastEthernet0/1
 description Interco vers ISR26 Ge0/0/1 - 90.0.26.0/24
 ip address 90.0.26.1 255.255.255.0
 ip nat outside
 ip access-group FILTRE-PRIVE out
 service-policy output QOS-VIDEO
 no shutdown
 exit
interface FastEthernet0/0
 description Trunk 802.1Q vers SW26-2 port 24
 no ip address
 service-policy input MARQUAGE
 service-policy output QOS-VIDEO
 no shutdown
 exit
interface FastEthernet0/0.150
 description VLAN 150 - Clients publics 150.150.26.0/24
 encapsulation dot1Q 150
 ip address 150.150.26.254 255.255.255.0
 exit
interface FastEthernet0/0.192
 description VLAN 192 - Clients prives 192.168.26.0/24
 encapsulation dot1Q 192
 ip address 192.168.26.254 255.255.255.0
 ip nat inside
 ip helper-address 50.50.26.1
 exit
interface Ethernet1/0
 description Serveurs publics 50.50.26.0/24 vers SW26-1 port 24
 ip address 50.50.26.254 255.255.255.0
 ip access-group PROTECT-SERVEURS out
 service-policy input MARQUAGE
 service-policy output QOS-VIDEO
 no shutdown
 exit
