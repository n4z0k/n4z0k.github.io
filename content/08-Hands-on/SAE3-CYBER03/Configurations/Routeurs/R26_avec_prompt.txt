R26(config)#enable secret Sae3Cyber!
R26(config)#username admin privilege 15 secret Sae3Cyber!
R26(config)#banner motd #Acces reserve - SAE3-CYBER03 - R26#
R26(config)#ip domain-name sae3cyber03.iut
!
! ---------- QoS (methode du TP4) : etape 1 - classification et marquage en entree ----------
! Flux video (TCP 8080) entre chaque serveur et les clients publics du site d'en face
R26(config)#ip cef
R26(config)#ip access-list extended VIDEO
R26(config-ext-nacl)#permit tcp host 50.50.25.1 eq 8080 150.150.26.0 0.0.0.255
R26(config-ext-nacl)#permit tcp host 50.50.26.1 eq 8080 150.150.25.0 0.0.0.255
R26(config-ext-nacl)#permit tcp 150.150.26.0 0.0.0.255 host 50.50.25.1 eq 8080
R26(config-ext-nacl)#permit tcp 150.150.25.0 0.0.0.255 host 50.50.26.1 eq 8080
R26(config-ext-nacl)#exit
R26(config)#class-map match-any VIDEO
R26(config-cmap)#match access-group name VIDEO
R26(config-cmap)#exit
R26(config)#policy-map MARQUAGE
R26(config-pmap)#class VIDEO
R26(config-pmap-c)#set dscp af41
R26(config-pmap-c)#exit
R26(config-pmap)#exit
!
! ---------- QoS (methode du TP4) : etape 2 - politique de sortie ----------
! Les flux video sont reconnus par leur marquage DSCP AF41 (34), pose a l'entree du reseau
R26(config)#class-map match-all DSCP-VIDEO
R26(config-cmap)#match dscp af41
R26(config-cmap)#exit
R26(config)#policy-map QOS-VIDEO
R26(config-pmap)#class DSCP-VIDEO
R26(config-pmap-c)#bandwidth percent 30
R26(config-pmap-c)#exit
R26(config-pmap)#class class-default
R26(config-pmap-c)#fair-queue
R26(config-pmap-c)#exit
R26(config-pmap)#exit
!
! ---------- ACL : protection du reseau des serveurs publics ----------
! Appliquee en SORTIE de l'interface serveurs : seul le trafic destine aux serveurs est examine
R26(config)#ip access-list extended PROTECT-SERVEURS
R26(config-ext-nacl)#remark Web
R26(config-ext-nacl)#permit tcp any host 50.50.26.1 eq www
R26(config-ext-nacl)#permit tcp any host 50.50.26.1 eq 443
R26(config-ext-nacl)#remark SSH
R26(config-ext-nacl)#permit tcp any host 50.50.26.1 eq 22
R26(config-ext-nacl)#remark FTP (controle, actif, passif 40000-40100)
R26(config-ext-nacl)#permit tcp any host 50.50.26.1 eq ftp
R26(config-ext-nacl)#permit tcp any host 50.50.26.1 eq ftp-data
R26(config-ext-nacl)#permit tcp any host 50.50.26.1 range 40000 40100
R26(config-ext-nacl)#remark Video : uniquement clients publics du site d'en face
R26(config-ext-nacl)#permit tcp 150.150.25.0 0.0.0.255 host 50.50.26.1 eq 8080
R26(config-ext-nacl)#remark RADIUS depuis le switch d'acces SW26-2 (NAS 802.1X)
R26(config-ext-nacl)#permit udp host 150.150.26.250 host 50.50.26.1 range 1812 1813
R26(config-ext-nacl)#remark DHCP relaye par le routeur
R26(config-ext-nacl)#permit udp any host 50.50.26.1 eq bootps
R26(config-ext-nacl)#remark ICMP pour les tests
R26(config-ext-nacl)#permit icmp any host 50.50.26.1 echo
R26(config-ext-nacl)#permit icmp any host 50.50.26.1 echo-reply
R26(config-ext-nacl)#permit icmp any host 50.50.26.1 unreachable
R26(config-ext-nacl)#deny ip any any log
R26(config-ext-nacl)#exit
!
! ---------- ACL : aucune adresse privee ne doit sortir vers l'ISR ----------
! Evaluee apres le NAT : seules les adresses deja traduites passent
R26(config)#ip access-list extended FILTRE-PRIVE
R26(config-ext-nacl)#deny ip 10.0.0.0 0.255.255.255 any
R26(config-ext-nacl)#deny ip 172.16.0.0 0.15.255.255 any
R26(config-ext-nacl)#deny ip 192.168.0.0 0.0.255.255 any
R26(config-ext-nacl)#permit ip any any
R26(config-ext-nacl)#exit
!
! ---------- NAT dynamique : 63 premieres adresses privees ----------
R26(config)#ip access-list standard NAT-PRIVE-26
R26(config-std-nacl)#permit 192.168.26.0 0.0.0.63
R26(config-std-nacl)#exit
R26(config)#ip nat pool POOL-NAT-26 90.0.26.2 90.0.26.64 netmask 255.255.255.0
R26(config)#ip nat inside source list NAT-PRIVE-26 pool POOL-NAT-26
!
! ---------- Interfaces ----------
R26(config)#interface FastEthernet0/1
R26(config-if)#description Interco vers ISR26 Ge0/0/1 - 90.0.26.0/24
R26(config-if)#ip address 90.0.26.1 255.255.255.0
R26(config-if)#ip nat outside
R26(config-if)#ip access-group FILTRE-PRIVE out
R26(config-if)#service-policy output QOS-VIDEO
R26(config-if)#no shutdown
R26(config-if)#exit
R26(config)#interface FastEthernet0/0
R26(config-if)#description Trunk 802.1Q vers SW26-2 port 24
R26(config-if)#no ip address
R26(config-if)#service-policy input MARQUAGE
R26(config-if)#service-policy output QOS-VIDEO
R26(config-if)#no shutdown
R26(config-if)#exit
R26(config)#interface FastEthernet0/0.150
R26(config-subif)#description VLAN 150 - Clients publics 150.150.26.0/24
R26(config-subif)#encapsulation dot1Q 150
R26(config-subif)#ip address 150.150.26.254 255.255.255.0
R26(config-subif)#exit
R26(config)#interface FastEthernet0/0.192
R26(config-subif)#description VLAN 192 - Clients prives 192.168.26.0/24
R26(config-subif)#encapsulation dot1Q 192
R26(config-subif)#ip address 192.168.26.254 255.255.255.0
R26(config-subif)#ip nat inside
R26(config-subif)#ip helper-address 50.50.26.1
R26(config-subif)#exit
R26(config)#interface Ethernet1/0
R26(config-if)#description Serveurs publics 50.50.26.0/24 vers SW26-1 port 24
R26(config-if)#ip address 50.50.26.254 255.255.255.0
R26(config-if)#ip access-group PROTECT-SERVEURS out
R26(config-if)#service-policy input MARQUAGE
R26(config-if)#service-policy output QOS-VIDEO
R26(config-if)#no shutdown
R26(config-if)#exit
